\begin{frame}[t, fragile]{Calling Convention}{}
	\vspace{\baselineskip}

	\begin{onlyenv}<1>
		\vspace{\baselineskip}
		\begin{itemize}
			\item \enquote{Aufrufkonvention} $\rightarrow$ lediglich eine Vereinbarung
			\item definiert Parameterübergabe, Rückgabe, Registersicherung, Stack etc.
			      \begin{enumerate}
				      \item Datentypen $\le$ 4 Byte in \texttt{a}-Register, sign-extension falls vorzeichenbehafteter Datentyp
				      \item Datentypen $=$ 8 Byte in zwei \texttt{a}-Registern, niedrigwertige Hälfte zuerst
				      \item Datentypen $>$ 8 Byte als Pointer auf Speicher (meist Caller-Stack)
				      \item Falls zu wenige Register: Übergabe über Stack
			      \end{enumerate}
			\item Stackpointer muss immer ein Vielfaches von 16 Byte sein!
		\end{itemize}
	\end{onlyenv}

	\begin{onlyenv}<2-3>
		\begin{minted}[escapeinside=||,beameroverlays=true, frame=single]{c}
			uint64_t foo(int32_t a, uint8_t b, uint64_t c);
		\end{minted}
	\end{onlyenv}

	\vspace{\baselineskip}

	\begin{center}
		\begin{onlyenv}<2-3>
			\begin{tabular}{ccc}
				Parameter & Übergabe untere 32 Bit                     & Übergabe obere 32 Bit     \\
				a         & \uncover<3->{\texttt{a0}}                  & \uncover<3->{---}         \\
				b         & \uncover<3->{\texttt{a1}  (zero-extended)} & \uncover<3->{---}         \\
				c         & \uncover<3->{\texttt{a2}}                  & \uncover<3->{\texttt{a3}} \\
				Rückgabe  & \uncover<3->{\texttt{a0}}                  & \uncover<3->{\texttt{a1}} \\
			\end{tabular}
		\end{onlyenv}
	\end{center}


\end{frame}

\begin{frame}[c]{Stack}{}
	\vspace*{-1cm}
	\begin{columns}[c]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Speicherbereich für lokale Variablen
				\item wächst von hohen Adressen zu niedrigen Adressen
				\item Platz auf dem Stack reservieren: \texttt{sp - x}
				\item Platz auf dem Stack freigeben: \texttt{sp + x}
				\item alles an Adressen \textless{} \texttt{sp} ungeschützt
				\item besonders wichtig für Rekursion (Stackframes)
			\end{itemize}

\begin{frame}[c, fragile]{Rekursion (2)}{}
	\begin{columns}[T]
		\begin{column}{0.3\textwidth}
			\begin{minted}{gas}
fun:
  addi sp, sp, -8
  sw ra, 0(sp)
  sw a0, 4(sp)
  li t0, 1
  beq a0, t0, end
  addi a0, a0, -1
  jal fun
  lw t0, 4(sp)
  mul a0, a0, t0
end:
  lw ra, 0(sp)
  addi sp, sp, 8
  ret
			\end{minted}
		\end{column}%
		\begin{column}{0.5\textwidth}
			\begin{onlyenv}<1>
				\begin{enumerate}
					\item Sicherung von \verb|ra| und evtl. Parametern
					\item Abbruchbedingung(en)
					\item Vorbereitung der Parameter für den rekursiven Aufruf
					\item Rekursiver Aufruf
					\item Ergebnis des Aufrufs verwerten
					\item Wiederherstellung von \verb|ra|, \verb|sp|
					\item Rücksprung
				\end{enumerate}
				\vspace*{\baselineskip}
				Schritte 3--5 können mehrmals vorkommen!

				\vspace*{\baselineskip}
				\small\textbf{Achtung}: 8 Byte nicht CC-konform, nur zur besseren Darstellung
			\end{onlyenv}
